cmake_minimum_required(VERSION 3.22)
project(OrTTSRuntime VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ORTTS_AUDIOCPP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/_work/audio.cpp" CACHE PATH "Pinned patched audio.cpp source")
if(NOT EXISTS "${ORTTS_AUDIOCPP_SOURCE}/src/models/qwen3_tts/ortts_joint/private_adapter.cpp")
  message(FATAL_ERROR "Run tools/prepare_runtime.py using the pinned local upstream checkout first")
endif()
foreach(flag ENGINE_ENABLE_CUDA ENGINE_ENABLE_HIP ENGINE_ENABLE_VULKAN ENGINE_ENABLE_METAL ENGINE_ENABLE_OPENMP ENGINE_ENABLE_LLAMAFILE ENGINE_ENABLE_NATIVE_CPU ENGINE_ENABLE_CPU_ALL_VARIANTS ENGINE_ENABLE_RKNN ENGINE_ENABLE_NEON ENGINE_BUILD_TESTS ENGINE_BUILD_EXAMPLES GGML_NATIVE GGML_OPENMP GGML_BACKEND_DL GGML_CPU_ALL_VARIANTS BUILD_TESTING)
  set(${flag} OFF CACHE BOOL "OrTTS CPU release baseline" FORCE)
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
set(AUDIOCPP_ORTTS_JOINT ON CACHE BOOL "OrTTS joint decoder adapter" FORCE)
set(AUDIOCPP_DEPLOYMENT_BUILD ON CACHE BOOL "Embed model contracts" FORCE)
set(AUDIOCPP_MODEL_SET custom CACHE STRING "Explicit runtime family" FORCE)
set(AUDIOCPP_MODELS qwen3_tts CACHE STRING "OrTTS Qwen frontend" FORCE)
set(AUDIOCPP_BUILD_C_API OFF CACHE BOOL "Link C API into standalone runner" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "Existing LFM decoder backend support" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "Keep measured baseline encoding" FORCE)
set(ORTTS_CPU_PROFILE "legacy" CACHE STRING "legacy, x86_64-baseline, x86_64-avx2, arm64-baseline, arm64-dotprod, arm64-i8mm")
if(NOT ORTTS_CPU_PROFILE STREQUAL "legacy")
  foreach(feature SSE42 AVX AVX2 AVX_VNNI BMI2 FMA F16C AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 AMX_TILE AMX_INT8 AMX_BF16)
    set(GGML_${feature} OFF CACHE BOOL "Explicit portable CPU profile" FORCE)
  endforeach()
  if(ORTTS_CPU_PROFILE STREQUAL "x86_64-avx2")
    foreach(feature SSE42 AVX AVX2 BMI2 FMA F16C)
      set(GGML_${feature} ON CACHE BOOL "AVX2 CPU profile" FORCE)
    endforeach()
  elseif(ORTTS_CPU_PROFILE STREQUAL "arm64-baseline")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "ARM baseline" FORCE)
  elseif(ORTTS_CPU_PROFILE STREQUAL "arm64-dotprod")
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "ARM DotProd" FORCE)
  elseif(ORTTS_CPU_PROFILE STREQUAL "arm64-i8mm")
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+i8mm+fp16" CACHE STRING "ARM i8mm" FORCE)
  elseif(NOT ORTTS_CPU_PROFILE STREQUAL "x86_64-baseline")
    message(FATAL_ERROR "Unknown ORTTS_CPU_PROFILE")
  endif()
endif()
# SentencePiece expects this helper when embedded under the Xcode generator.
if(APPLE AND NOT COMMAND set_xcode_property)
 function(set_xcode_property target property value)
  set_property(TARGET ${target} PROPERTY XCODE_ATTRIBUTE_${property} "${value}")
 endfunction()
endif()
add_subdirectory("${ORTTS_AUDIOCPP_SOURCE}" audio EXCLUDE_FROM_ALL)
add_executable(ortts-tts src/tts_main.cpp "${ORTTS_AUDIOCPP_SOURCE}/src/capi/audiocpp.cpp")
target_include_directories(ortts-tts PRIVATE "${ORTTS_AUDIOCPP_SOURCE}/include")
target_compile_definitions(ortts-tts PRIVATE AUDIOCPP_ORTTS_JOINT=1 AUDIOCPP_VERSION_STRING="ortts-demo-local-0.1.0")
target_link_libraries(ortts-tts PRIVATE engine_runtime ggml)
if(ANDROID)
  target_link_options(ortts-tts PRIVATE "-Wl,-z,max-page-size=16384")
endif()
install(TARGETS ortts-tts RUNTIME DESTINATION bin BUNDLE DESTINATION bin)

option(ORTTS_BUILD_ENGINE_PLUGIN "Build embeddable CPU engine C ABI" OFF)
if(ORTTS_BUILD_ENGINE_PLUGIN)
 if(ORTTS_CPU_PROFILE STREQUAL "legacy")
   message(FATAL_ERROR "Plugin builds require an explicit ORTTS_CPU_PROFILE; use tools/build_engine.py")
 endif()
 add_library(ortts_engine SHARED runtime/include/ortts_engine.h runtime/src/ortts_engine.cpp "${ORTTS_AUDIOCPP_SOURCE}/src/capi/audiocpp.cpp")
 target_include_directories(ortts_engine PUBLIC runtime/include PRIVATE "${ORTTS_AUDIOCPP_SOURCE}/include")
 target_compile_definitions(ortts_engine PRIVATE ORTTS_ENGINE_BUILD AUDIOCPP_ORTTS_JOINT=1 AUDIOCPP_VERSION_STRING="ortts-cpu-engine-preview")
 target_link_libraries(ortts_engine PRIVATE engine_runtime ggml)
 set_target_properties(ortts_engine PROPERTIES OUTPUT_NAME ortts CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
 if(APPLE)
   target_link_options(ortts_engine PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/runtime/exports.apple")
   if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
     set_target_properties(ortts_engine PROPERTIES FRAMEWORK TRUE OUTPUT_NAME OrTTS PUBLIC_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/runtime/include/ortts_engine.h" MACOSX_FRAMEWORK_IDENTIFIER com.orient.ortts.engine XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO XCODE_ATTRIBUTE_DEFINES_MODULE YES INSTALL_NAME_DIR "@rpath")
     configure_file(runtime/module.modulemap.in "${CMAKE_CURRENT_BINARY_DIR}/module.modulemap" COPYONLY)
     target_sources(ortts_engine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/module.modulemap")
     set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/module.modulemap" PROPERTIES MACOSX_PACKAGE_LOCATION Modules HEADER_FILE_ONLY TRUE)
   endif()
 elseif(UNIX)
   target_link_options(ortts_engine PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/runtime/exports.map" "-Wl,--exclude-libs,ALL")
 endif()
 if(ANDROID)
   target_link_options(ortts_engine PRIVATE "-Wl,-z,max-page-size=16384")
 endif()
 install(TARGETS ortts_engine LIBRARY DESTINATION lib COMPONENT Engine FRAMEWORK DESTINATION . COMPONENT Engine)
 install(FILES runtime/include/ortts_engine.h DESTINATION include COMPONENT Engine)
endif()

option(ORTTS_BUILD_ANDROID_JNI "Build Android app libraries" OFF)
if(ORTTS_BUILD_ANDROID_JNI)
 if(NOT ANDROID)
  message(FATAL_ERROR "Android JNI requires an Android toolchain")
 endif()
 add_library(ortts_capi SHARED "${ORTTS_AUDIOCPP_SOURCE}/src/capi/audiocpp.cpp")
 set_target_properties(ortts_capi PROPERTIES OUTPUT_NAME audiocpp)
 target_include_directories(ortts_capi PRIVATE "${ORTTS_AUDIOCPP_SOURCE}/include")
 target_compile_definitions(ortts_capi PRIVATE AUDIOCPP_ORTTS_JOINT=1 AUDIOCPP_VERSION_STRING="ortts-0.2.0-preview")
 target_link_libraries(ortts_capi PRIVATE engine_runtime ggml)
 target_link_options(ortts_capi PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--version-script=${ORTTS_AUDIOCPP_SOURCE}/src/capi/audiocpp.map")
 add_library(ortts_jni SHARED android/app/src/main/cpp/ortts_jni.cpp)
 target_include_directories(ortts_jni PRIVATE "${ORTTS_AUDIOCPP_SOURCE}/include")
 target_link_libraries(ortts_jni PRIVATE ortts_capi log)
 target_link_options(ortts_jni PRIVATE "-Wl,-z,max-page-size=16384")
 add_library(ortts_device SHARED android/app/src/main/cpp/device_support.c)
 target_compile_options(ortts_device PRIVATE -march=armv8-a)
 target_link_options(ortts_device PRIVATE "-Wl,-z,max-page-size=16384")
endif()
